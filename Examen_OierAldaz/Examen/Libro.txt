public class Libro
{
    String titulo;
    String autor;
    String ISBN;
    Boolean prestado;
}

public String getAutor()
{
    return autor;
}

public void setAutor(String autor)
{
    this.autor = autor;
}

public String toString()
{
    return "Libro [titulo=" + titulo + ", autor" + autor + ", ISBN=" + ISBN + "; prestado=" + prestado + "]";
}
